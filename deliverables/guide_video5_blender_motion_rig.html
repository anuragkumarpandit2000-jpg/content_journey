4. The Blender Python OSC Receiver & Live Recorder

Copy and run this script inside Blender's Scripting Workspace. It binds to your PC's local WiFi IP and live-drives the active camera: 

import bpy
import socket
import struct
import math

class SmartphoneMotionRig(bpy.types.Operator):
    bl_idname = "wm.smartphone_motion_rig"
    bl_label = "Connect Smartphone Camera Rig"
    
    _timer = None
    _sock = None
    
    def modal(self, context, event):
        if event.type in {'RIGHTMOUSE', 'ESC'}:
            self.cancel(context)
            return {'CANCELLED'}
            
        if event.type == 'TIMER':
            cam = context.scene.camera
            if not cam:
                return {'PASS_THROUGH'}
                
            try:
                data, _ = self._sock.recvfrom(1024)
                # Parse OSC orientation packet (Pitch, Yaw, Roll)
                if len(data) >= 16:
                    pitch, yaw, roll = struct.unpack("!fff", data[4:16])
                    cam.rotation_euler = (
                        math.radians(pitch),
                        math.radians(roll),
                        math.radians(yaw)
                    )
                    # Automatically insert keyframe if timeline is playing
                    if context.screen.is_animation_playing:
                        cam.keyframe_insert(data_path="rotation_euler")
            except BlockingIOError:
                pass
                
        return {'PASS_THROUGH'}

    def execute(self, context):
        self._sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self._sock.bind(("0.0.0.0", 8000))
        self._sock.setblocking(False)
        
        wm = context.window_manager
        self._timer = wm.event_timer_add(1.0 / 60.0, window=context.window)
        wm.modal_handler_add(self)
        self.report({'INFO'}, "Camera Rig Connected on Port 8000! Press Space to Record, ESC to Stop.")
        return {'RUNNING_MODAL'}

    def cancel(self, context):
        wm = context.window_manager
        wm.event_timer_remove(self._timer)
        if self._sock:
            self._sock.close()
        self.report({'INFO'}, "Smartphone Camera Rig Disconnected.")

bpy.utils.register_class(SmartphoneMotionRig)
bpy.ops.wm.smartphone_motion_rig()

5. Step-by-Step Production Workflow

  1. Connect your smartphone and workstation PC to the same 5GHz WiFi network (or create a mobile hotspot for lowest latency).
  2. Open CMD on Windows and type ipconfig to find your IPv4 Address (e.g. 192.168.1.15).
  3. In the mobile app, set target host IP to your PC's IP and target port to 8000.
  4. In Blender, select your Camera, switch to Camera View (Numpad 0), and run the script above.
  5. Move your phone: your Blender 3D Camera now tracks your real physical movements in real-time!
  6. Press Spacebar in Blender to play timeline: every subtle shake, tilt, and swoop is baked frame-by-frame into your Action timeline at 60 FPS.
