2. Master FFmpeg Command Line

ffmpeg -i "input_video.mp4" \
  -vf "scale=1080:1920:force_original_aspect_ratio=decrease,pad=1080:1920:(ow-iw)/2:(oh-ih)/2,format=yuv420p" \
  -c:v libx264 \
  -preset slow \
  -profile:v high \
  -level 4.2 \
  -crf 23 \
  -maxrate 15M \
  -bufsize 30M \
  -colorspace bt709 \
  -color_primaries bt709 \
  -color_trc bt709 \
  -c:a aac \
  -b:a 320k \
  -ar 48000 \
  -movflags +faststart \
  "instagram_master_sharp.mp4"
